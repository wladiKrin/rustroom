```rust
//! Generic undo/redo history over snapshots of a value.

/// Holds a current value plus undo/redo stacks of prior snapshots.
pub struct History<T> {
    current: T,
    undo: Vec<T>,
    redo: Vec<T>,
}

impl<T: Clone + PartialEq> History<T> {
    pub fn new(initial: T) -> Self {
        Self {
            current: initial,
            undo: Vec::new(),
            redo: Vec::new(),
        }
    }

    pub fn get(&self) -> &T {
        &self.current
    }

    /// Record `value` as a new distinct step. No-op if equal to the current value.
    /// Clears the redo stack.
    pub fn push(&mut self, value: T) {
        if value != self.current {
            self.undo.push(self.current.clone());
            self.current = value;
            self.redo.clear();
        }
    }

    pub fn can_undo(&self) -> bool {
        !self.undo.is_empty()
    }

    pub fn can_redo(&self) -> bool {
        !self.redo.is_empty()
    }

    /// Step back. Returns true if something changed.
    pub fn undo(&mut self) -> bool {
        if let Some(prev) = self.undo.pop() {
            self.redo.push(std::mem::replace(&mut self.current, prev));
            true
        } else {
            false
        }
    }

    /// Step forward. Returns true if something changed.
    pub fn redo(&mut self) -> bool {
        if let Some(next) = self.redo.pop() {
            self.undo.push(std::mem::replace(&mut self.current, next));
            true
        } else {
            false
        }
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn push_undo_redo_cycle() {
        let mut h = History::new(0i32);
        h.push(1);
        h.push(2);
        assert_eq!(*h.get(), 2);
        assert!(h.undo());
        assert_eq!(*h.get(), 1);
        assert!(h.undo());
        assert_eq!(*h.get(), 0);
        assert!(!h.undo()); // nothing left
        assert!(h.redo());
        assert_eq!(*h.get(), 1);
    }

    #[test]
    fn push_equal_is_noop() {
        let mut h = History::new(5i32);
        h.push(5);
        assert!(!h.can_undo());
    }

    #[test]
    fn push_clears_redo() {
        let mut h = History::new(0i32);
        h.push(1);
        h.undo();
        assert!(h.can_redo());
        h.push(9); // new branch
        assert!(!h.can_redo());
        assert_eq!(*h.get(), 9);
    }
}
```

```rust
/// Render `edits` against `buffer` at a working resolution capped by `max_dim`.
/// Pure and synchronous — used by the worker and for one-off renders (e.g. baseline).
pub fn render_once(buffer: &ImageBuf, edits: &Edits, max_dim: usize) -> RenderedFrame {
    let preview = buffer.downscale_to(max_dim);
    let out = build_pipeline(edits).run(preview);
    let (width, height) = (out.width(), out.height());
    let mut rgba = Vec::with_capacity(width * height * 4);
    for px in out.data().chunks_exact(3) {
        rgba.push((px[0].clamp(0.0, 1.0) * 255.0).round() as u8);
        rgba.push((px[1].clamp(0.0, 1.0) * 255.0).round() as u8);
        rgba.push((px[2].clamp(0.0, 1.0) * 255.0).round() as u8);
        rgba.push(255);
    }
    RenderedFrame { width, height, rgba }
}
```

```rust
//! Pure crop-rectangle interaction math on `edit_model::CropRect` (normalized [0,1]).

use edit_model::CropRect;

const MIN_SIZE: f32 = 0.05; // minimum normalized crop extent

/// Which part of the crop rectangle a point is over.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Handle {
    TopLeft,
    TopRight,
    BottomLeft,
    BottomRight,
    Left,
    Right,
    Top,
    Bottom,
    Inside,
}

/// Which handle (if any) is under normalized point `p`, within `tol` (normalized).
pub fn hit_test(r: &CropRect, p: [f32; 2], tol: f32) -> Option<Handle> {
    let (x, y) = (p[0], p[1]);
    let near = |a: f32, b: f32| (a - b).abs() <= tol;
    let in_x = x >= r.left - tol && x <= r.right + tol;
    let in_y = y >= r.top - tol && y <= r.bottom + tol;
    // Corners first.
    if near(x, r.left) && near(y, r.top) {
        return Some(Handle::TopLeft);
    }
    if near(x, r.right) && near(y, r.top) {
        return Some(Handle::TopRight);
    }
    if near(x, r.left) && near(y, r.bottom) {
        return Some(Handle::BottomLeft);
    }
    if near(x, r.right) && near(y, r.bottom) {
        return Some(Handle::BottomRight);
    }
    // Edges.
    if near(x, r.left) && in_y {
        return Some(Handle::Left);
    }
    if near(x, r.right) && in_y {
        return Some(Handle::Right);
    }
    if near(y, r.top) && in_x {
        return Some(Handle::Top);
    }
    if near(y, r.bottom) && in_x {
        return Some(Handle::Bottom);
    }
    // Inside.
    if x > r.left && x < r.right && y > r.top && y < r.bottom {
        return Some(Handle::Inside);
    }
    None
}

/// Apply a normalized drag `(dx, dy)` to `handle`, clamped to [0,1] with a minimum size.
pub fn apply_drag(r: CropRect, handle: Handle, dx: f32, dy: f32) -> CropRect {
    let mut out = r;
    match handle {
        Handle::Inside => {
            // Move whole rect, clamped so it stays inside [0,1].
            let w = out.right - out.left;
            let h = out.bottom - out.top;
            let nl = (out.left + dx).clamp(0.0, 1.0 - w);
            let nt = (out.top + dy).clamp(0.0, 1.0 - h);
            out.left = nl;
            out.right = nl + w;
            out.top = nt;
            out.bottom = nt + h;
        }
        _ => {
            if matches!(handle, Handle::Left | Handle::TopLeft | Handle::BottomLeft) {
                out.left = (out.left + dx).clamp(0.0, out.right - MIN_SIZE);
            }
            if matches!(handle, Handle::Right | Handle::TopRight | Handle::BottomRight) {
                out.right = (out.right + dx).clamp(out.left + MIN_SIZE, 1.0);
            }
            if matches!(handle, Handle::Top | Handle::TopLeft | Handle::TopRight) {
                out.top = (out.top + dy).clamp(0.0, out.bottom - MIN_SIZE);
            }
            if matches!(handle, Handle::Bottom | Handle::BottomLeft | Handle::BottomRight) {
                out.bottom = (out.bottom + dy).clamp(out.top + MIN_SIZE, 1.0);
            }
        }
    }
    out
}

/// Snap to a normalized aspect (width/height in normalized units), centered and maximal.
/// `norm_aspect` = pixel_aspect / image_aspect, where image_aspect = image_w / image_h.
pub fn snap_to_aspect(norm_aspect: f32) -> CropRect {
    // Largest centered rect of the given normalized aspect that fits in the unit square.
    let (w, h) = if norm_aspect >= 1.0 {
        (1.0, 1.0 / norm_aspect)
    } else {
        (norm_aspect, 1.0)
    };
    CropRect {
        left: (1.0 - w) / 2.0,
        top: (1.0 - h) / 2.0,
        right: (1.0 + w) / 2.0,
        bottom: (1.0 + h) / 2.0,
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    fn r() -> CropRect {
        CropRect { left: 0.2, top: 0.2, right: 0.8, bottom: 0.8 }
    }

    #[test]
    fn hit_test_corner_and_inside() {
        assert_eq!(hit_test(&r(), [0.2, 0.2], 0.03), Some(Handle::TopLeft));
        assert_eq!(hit_test(&r(), [0.5, 0.5], 0.03), Some(Handle::Inside));
        assert_eq!(hit_test(&r(), [0.05, 0.05], 0.03), None);
    }

    #[test]
    fn drag_right_edge_grows() {
        let out = apply_drag(r(), Handle::Right, 0.1, 0.0);
        assert!((out.right - 0.9).abs() < 1e-6);
        assert!((out.left - 0.2).abs() < 1e-6);
    }

    #[test]
    fn drag_inside_clamps_within_unit() {
        let out = apply_drag(r(), Handle::Inside, 0.5, 0.0); // would overflow right
        assert!(out.right <= 1.0 + 1e-6);
        assert!((out.right - out.left - 0.6).abs() < 1e-6); // width preserved
    }

    #[test]
    fn min_size_enforced() {
        let out = apply_drag(r(), Handle::Left, 1.0, 0.0); // drag left edge far right
        assert!(out.right - out.left >= MIN_SIZE - 1e-6);
    }

    #[test]
    fn snap_square_is_centered() {
        let sq = snap_to_aspect(1.0);
        assert!((sq.left - 0.0).abs() < 1e-6 && (sq.right - 1.0).abs() < 1e-6);
        let wide = snap_to_aspect(2.0); // twice as wide as tall
        assert!((wide.bottom - wide.top - 0.5).abs() < 1e-6);
        assert!((wide.right - wide.left - 1.0).abs() < 1e-6);
    }
}
```
