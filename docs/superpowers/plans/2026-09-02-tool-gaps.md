```rust
/// Floor of the halo suppressor: what fraction of Clarity survives at a hard edge. (`s` in the spec.)
pub(crate) const CLARITY_HALO_FLOOR: f32 = 0.25;
/// Coarse-band excursion above which the suppressor engages. Derived, not copied from sharpening's 0.05 — see Step 3.
pub(crate) const CLARITY_HALO_KNEE: f32 = <derived>;

fn halo_suppression(d: f32, floor: f32) -> f32 {
    floor + (1.0 - floor) / (1.0 + (d.abs() / CLARITY_HALO_KNEE).powi(2))
}
```

`apply` delegates to `pub(crate) fn apply_with(&self, image, halo_floor: f32)`; the clarity accumulation becomes `k_c * (lv - b) * midtone_weight(lv) * halo_suppression(lv - b, halo_floor)`.
