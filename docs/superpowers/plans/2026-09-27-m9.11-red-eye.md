```rust
/// One red- or pet-eye correction: a circle around a pupil, detected at the click and stored.
///
/// ⚠ **The stage never detects** (M9.11 spec, approach A): an eye that re-detected on every render
/// would move whenever an upstream edit changed the pixels. The circle is found once, in the app,
/// and from then on it is an edit like any other.
///
/// ⚠ Same space as [`Spot`]: post-lens, pre-perspective frame, normalised per axis; `radius` a
/// fraction of the long edge. Same `Copy` and no container-level `serde(default)`, for `Spot`'s
/// reasons.
#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
pub struct Eye {
    pub centre: [f32; 2],
    pub radius: f32,
    pub kind: EyeKind,
    /// 0..=100: how far the correction reaches into weakly red (pet: dimmer) pixels. The circle
    /// still bounds it.
    pub pupil_size: f32,
    /// 0..=100: how dark the corrected pupil ends up.
    pub darken: f32,
    /// Pet only: the catchlight's offset from `centre`, in **radius** units of the stage's frame.
    /// `None` = no catchlight. ⚠ Stored, not derived in the stage: the stage's frame is
    /// pre-orientation, so "up-left on screen" is only knowable in the app.
    #[serde(default)]
    pub catchlight: Option<[f32; 2]>,
}

/// What kind of flash reflection an [`Eye`] corrects.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, Default)]
pub enum EyeKind {
    /// A human red pupil: the red is removed and the pupil darkened.
    #[default]
    Red,
    /// An animal's green, yellow or white eyeshine: replaced by a dark pupil.
    Pet,
}
```

```rust
//! Red- and pet-eye correction: each stored [`Eye`] circle recolours the pupil pixels inside it.
//!
//! ⚠⚠ **No detection here** (M9.11, approach A). The circle was found once, at the click; this
//! stage only corrects inside it, per pixel, in proportion to how red (pet: how bright) each pixel
//! is — so skin and iris caught inside a slightly generous circle survive.
//!
//! ⚠ A **prefix** step right after spot removal, in the same post-lens, pre-perspective space, so
//! `SpotMap` draws it and a drag re-runs only this and the geometry. No GPU twin (like spots).

use crate::stages::spot_removal::coverage;
use crate::PipelineStage;
use core_image::ImageBuf;
use edit_model::{Eye, EyeKind};

/// Feather of the circle's own edge, `Spot::feather`'s 0..=100 scale.
const EDGE_FEATHER: f32 = 30.0;
/// Width of the redness / brightness ramp above its threshold.
const RAMP: f32 = 0.12;
/// Catchlight disc radius, in eye radii.
const CATCHLIGHT_R: f32 = 0.18;

/// How red a linear ProPhoto pixel is: `(R − max(G,B)) / R`, in `[0, 1]`.
///
/// ⚠ **One definition, used by detection too** (`app::eye_detect`). Chosen by measurement over
/// `(R − (G+B)/2)/R`, which put skin at 0.27–0.47 and so overlapped flash pupils; this one puts
/// skin at ≤ 0.37 and pupils at ≥ 0.56 — `redness_separates_skin_from_flash_pupils`.
pub fn redness(p: [f32; 3]) -> f32 {
    ((p[0] - p[1].max(p[2])) / p[0].max(1e-4)).clamp(0.0, 1.0)
}

fn smoothstep(e0: f32, e1: f32, x: f32) -> f32 {
    let t = ((x - e0) / (e1 - e0)).clamp(0.0, 1.0);
    t * t * (3.0 - 2.0 * t)
}

fn lerp(a: f32, b: f32, t: f32) -> f32 {
    a + (b - a) * t
}

pub struct RedEyeStage {
    eyes: Vec<Eye>,
}

impl RedEyeStage {
    pub fn new(eyes: &[Eye]) -> Self {
        Self { eyes: eyes.to_vec() }
    }
}

impl PipelineStage for RedEyeStage {
    fn name(&self) -> &'static str {
        "red_eye"
    }

    fn apply(&self, image: ImageBuf) -> ImageBuf {
        let mut image = image;
        for e in &self.eyes {
            apply_eye(&mut image, e);
        }
        image
    }
}

fn apply_eye(image: &mut ImageBuf, e: &Eye) {
    let (w, h) = (image.width(), image.height());
    if w == 0 || h == 0 {
        return;
    }
    let r = e.radius * w.max(h) as f32;
    if !r.is_finite() || r <= 0.0 {
        return;
    }
    let cx = e.centre[0] * (w - 1) as f32;
    let cy = e.centre[1] * (h - 1) as f32;
    let x0 = (cx - r).floor().max(0.0) as usize;
    let y0 = (cy - r).floor().max(0.0) as usize;
    let x1 = ((cx + r).ceil() as usize).min(w - 1);
    let y1 = ((cy + r).ceil() as usize).min(h - 1);
    if (cx + r) < 0.0 || (cy + r) < 0.0 || x0 > x1 || y0 > y1 {
        return;
    }
    let size = (e.pupil_size / 100.0).clamp(0.0, 1.0);
    let dark = (e.darken / 100.0).clamp(0.0, 1.0);
    let disc = |x: usize, y: usize| ((x as f32 - cx).powi(2) + (y as f32 - cy).powi(2)).sqrt();

    match e.kind {
        EyeKind::Red => {
            let t0 = lerp(0.60, 0.40, size);
            let k = lerp(1.0, 0.25, dark);
            for y in y0..=y1 {
                for x in x0..=x1 {
                    let c = coverage(disc(x, y), r, EDGE_FEATHER);
                    if c <= 0.0 {
                        continue;
                    }
                    let p = image.pixel(x, y);
                    let wgt = c * smoothstep(t0, t0 + RAMP, redness(p));
                    if wgt <= 0.0 {
                        continue;
                    }
                    let fixed = [0.5 * (p[1] + p[2]) * k, p[1] * k, p[2] * k];
                    image.set_pixel(x, y, std::array::from_fn(|i| lerp(p[i], fixed[i], wgt)));
                }
            }
        }
        EyeKind::Pet => {
            // The 95th-percentile luminance inside the circle: the eyeshine's own level, so the
            // threshold follows the exposure rather than an absolute number.
            let mut ys: Vec<f32> = Vec::new();
            for y in y0..=y1 {
                for x in x0..=x1 {
                    if disc(x, y) <= r {
                        ys.push(color::luminance(image.pixel(x, y)));
                    }
                }
            }
            if ys.is_empty() {
                return;
            }
            ys.sort_by(|a, b| a.total_cmp(b));
            let y_ref = ys[((ys.len() - 1) as f32 * 0.95) as usize].max(1e-6);
            let t0 = lerp(0.70, 0.20, size);
            let k = lerp(0.15, 0.03, dark);
            let cl = e.catchlight.map(|o| (cx + o[0] * r, cy + o[1] * r));
            for y in y0..=y1 {
                for x in x0..=x1 {
                    let c = coverage(disc(x, y), r, EDGE_FEATHER);
                    if c <= 0.0 {
                        continue;
                    }
                    let p = image.pixel(x, y);
                    let lum = color::luminance(p);
                    let wgt = c * smoothstep(t0, t0 + RAMP, lum / y_ref);
                    let mut fixed = [lum * k; 3];
                    if let Some((lx, ly)) = cl {
                        let d = ((x as f32 - lx).powi(2) + (y as f32 - ly).powi(2)).sqrt();
                        let a = coverage(d, CATCHLIGHT_R * r, 60.0);
                        fixed = fixed.map(|v| lerp(v, 0.9 * y_ref, a));
                    }
                    if wgt <= 0.0 {
                        continue;
                    }
                    image.set_pixel(x, y, std::array::from_fn(|i| lerp(p[i], fixed[i], wgt)));
                }
            }
        }
    }
}
```

```rust
//! Finding a pupil around a click — **once**, at the click (M9.11, approach A).
//!
//! ⚠ No `ImageBuf`: the search reads an injected sampler over the red-eye stage's input space —
//! `spot_source`'s design, and the same sampler (`spot_overlay::sampler_for`) feeds it.
//! ⚠ No egui: testable headless, like every lib module.

use edit_model::EyeKind;

/// Grid cells per side of the search window.
pub const GRID: usize = 64;
/// The window's half-width, as a fraction of the long edge.
pub const WINDOW: f32 = 0.03;
/// A seed below this redness is not a red pupil (skin tops out ~0.37, pupils start ~0.56).
pub const RED_FLOOR: f32 = 0.50;
/// The fill threshold never drops below this, so the blob cannot leak into flushed skin.
pub const RED_FILL_MIN: f32 = 0.42;
/// A blob whose bounding box is longer than this against its width is not a pupil.
pub const MAX_ELONGATION: f32 = 2.5;
/// Padding from the blob's equal-area radius to the stored circle.
pub const PAD: f32 = 1.2;

#[derive(Debug, Clone, Copy, PartialEq)]
pub enum Detection {
    Found { centre: [f32; 2], radius: f32 },
    NotFound,
}

/// Search around `click` (normalised) in a frame of `dims` pixels.
pub fn detect(
    click: [f32; 2],
    kind: EyeKind,
    dims: (usize, usize),
    sample: &dyn Fn([f32; 2]) -> [f32; 3],
) -> Detection {
    let (w, h) = (dims.0.max(1) as f32, dims.1.max(1) as f32);
    let long = w.max(h);
    let half = WINDOW * long; // pixels
    let step = 2.0 * half / GRID as f32;
    let at = |i: usize, j: usize| -> [f32; 2] {
        let x = click[0] * w - half + (i as f32 + 0.5) * step;
        let y = click[1] * h - half + (j as f32 + 0.5) * step;
        [x / w, y / h]
    };
    let score: Vec<f32> = (0..GRID * GRID)
        .map(|k| {
            let p = sample(at(k % GRID, k / GRID));
            match kind {
                EyeKind::Red => pipeline::redness(p),
                EyeKind::Pet => color::luminance(p),
            }
        })
        .collect();

    let (floor, fill_min) = match kind {
        EyeKind::Red => (RED_FLOOR, RED_FILL_MIN),
        EyeKind::Pet => {
            let mut sorted = score.clone();
            sorted.sort_by(|a, b| a.total_cmp(b));
            let median = sorted[sorted.len() / 2];
            (2.0 * median.max(1e-4), 2.0 * median.max(1e-4))
        }
    };

    // Seed: the best cell in the central quarter (half the window each way).
    let (lo, hi) = (GRID / 4, 3 * GRID / 4);
    let seed = (lo..hi)
        .flat_map(|j| (lo..hi).map(move |i| (i, j)))
        .max_by(|a, b| score[a.1 * GRID + a.0].total_cmp(&score[b.1 * GRID + b.0]))
        .expect("a non-empty centre");
    let seed_score = score[seed.1 * GRID + seed.0];
    if seed_score < floor {
        return Detection::NotFound;
    }
    let factor = match kind {
        EyeKind::Red => 0.6,
        EyeKind::Pet => 0.5,
    };
    let thr = fill_min.max(factor * seed_score);

    // 8-connected flood fill over the grid.
    let mut seen = vec![false; GRID * GRID];
    let mut stack = vec![seed];
    seen[seed.1 * GRID + seed.0] = true;
    let (mut n, mut sx, mut sy) = (0usize, 0.0f32, 0.0f32);
    let (mut minx, mut maxx, mut miny, mut maxy) = (GRID, 0, GRID, 0);
    while let Some((i, j)) = stack.pop() {
        if i == 0 || j == 0 || i == GRID - 1 || j == GRID - 1 {
            return Detection::NotFound; // touches the window edge: not a pupil-sized blob
        }
        n += 1;
        sx += i as f32;
        sy += j as f32;
        (minx, maxx, miny, maxy) = (minx.min(i), maxx.max(i), miny.min(j), maxy.max(j));
        for dj in -1i32..=1 {
            for di in -1i32..=1 {
                let (ni, nj) = ((i as i32 + di) as usize, (j as i32 + dj) as usize);
                let k = nj * GRID + ni;
                if !seen[k] && score[k] >= thr {
                    seen[k] = true;
                    stack.push((ni, nj));
                }
            }
        }
    }
    let (bw, bh) = ((maxx - minx + 1) as f32, (maxy - miny + 1) as f32);
    if bw.max(bh) / bw.min(bh) > MAX_ELONGATION {
        return Detection::NotFound;
    }
    let centre = at(0, 0);
    let c = [
        centre[0] + (sx / n as f32) * step / w,
        centre[1] + (sy / n as f32) * step / h,
    ];
    let r_px = PAD * (n as f32 * step * step / std::f32::consts::PI).sqrt();
    Detection::Found {
        centre: c,
        radius: (r_px / long).clamp(crate::spot_tool::MIN_RADIUS, crate::spot_tool::MAX_RADIUS),
    }
}
```

(`at(0,0)` is the centre of cell (0,0); adding `mean_index * step` in pixels gives the blob centroid. `i`/`j` never underflow in the neighbour loop because edge cells return before expanding. Check `app`'s `Cargo.toml` has `color` and `pipeline`; add `color` if missing.)

```rust
//! The red-eye tool's settings and the logic behind its gestures — no egui (`spot_tool`'s shape).

use crate::eye_detect::Detection;
use crate::spot_tool::{MAX_RADIUS, MIN_RADIUS};
use edit_model::{Eye, EyeKind};

/// Distance of a new pet eye's catchlight from the centre, in eye radii.
pub const CATCHLIGHT_OFFSET: f32 = 0.35;
/// The rim band a resize grab accepts, as a fraction of the radius either side of the rim.
pub const RIM_BAND: f32 = 0.25;

/// Live settings, session-lifetime (`SpotTool`'s rule). `active` is the mode: `None` means the
/// Retouch tool is placing patches (Heal/Clone), `Some(kind)` means it is placing eyes.
#[derive(Clone, Copy, Debug, PartialEq)]
pub struct EyeTool {
    pub active: Option<EyeKind>,
    /// The radius a `NotFound` click places — and a drag writes back.
    pub radius: f32,
    pub pupil_size: f32,
    pub darken: f32,
    /// Whether a new pet eye gets a catchlight.
    pub catchlight: bool,
}

impl Default for EyeTool {
    fn default() -> Self {
        Self { active: None, radius: 0.006, pupil_size: 50.0, darken: 50.0, catchlight: true }
    }
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Grab {
    Move,
    Resize,
}

/// A new eye from a detection (or the click, when nothing was found).
pub fn place(found: Detection, click: [f32; 2], kind: EyeKind, t: &EyeTool, catchlight: Option<[f32; 2]>) -> Eye {
    let (centre, radius) = match found {
        Detection::Found { centre, radius } => (centre, radius),
        Detection::NotFound => (click, t.radius),
    };
    Eye {
        centre,
        radius: radius.clamp(MIN_RADIUS, MAX_RADIUS),
        kind,
        pupil_size: t.pupil_size,
        darken: t.darken,
        catchlight: if kind == EyeKind::Pet { catchlight } else { None },
    }
}

/// Which eye (last first) and which grab `p` lands on. Distances in long-edge units, as
/// `spot_tool::hit_test`; `min_grab` keeps a tiny eye grabbable.
pub fn hit_test(eyes: &[Eye], p: [f32; 2], aspect: f32, min_grab: f32) -> Option<(usize, Grab)> {
    let (sx, sy) = crate::spot_tool::long_edge_scale(aspect);
    eyes.iter().enumerate().rev().find_map(|(i, e)| {
        let d = ((p[0] - e.centre[0]) * sx).hypot((p[1] - e.centre[1]) * sy);
        let band = (RIM_BAND * e.radius).max(min_grab * 0.5);
        if (d - e.radius).abs() <= band {
            Some((i, Grab::Resize))
        } else if d < e.radius {
            Some((i, Grab::Move))
        } else {
            None
        }
    })
}

/// The catchlight offset (radius units, length [`CATCHLIGHT_OFFSET`]) pointing from `centre`
/// toward `toward`, both normalised in a frame of `dims` pixels. The overlay passes the source
/// point under "up-left of the centre on screen", which is what makes the default orientation-proof.
pub fn catchlight_offset(centre: [f32; 2], toward: [f32; 2], dims: (usize, usize)) -> [f32; 2] {
    let (dx, dy) = ((toward[0] - centre[0]) * dims.0 as f32, (toward[1] - centre[1]) * dims.1 as f32);
    let len = dx.hypot(dy);
    if len <= f32::EPSILON {
        return [-CATCHLIGHT_OFFSET / 2f32.sqrt(); 2];
    }
    [CATCHLIGHT_OFFSET * dx / len, CATCHLIGHT_OFFSET * dy / len]
}
```

```rust
    };
    let mut changed = false;

    if let Some(kind) = tool.active {
        // ---- click: select, or detect and place ----
        let click = ui
            .interact(img_rect, ui.id().with("red_eye_click"), egui::Sense::click())
            .on_hover_cursor(egui::CursorIcon::Crosshair);
        if click.clicked() {
            if let Some(p) = click.interact_pointer_pos() {
                let at = to_source(p);
                match eye_tool::hit_test(&s.edits.eyes, at, aspect, min_grab) {
                    Some((i, _)) => select(s, tool, i),
                    None => {
                        let found = eye_detect::detect(at, kind, dims, sample);
                        if found == Detection::NotFound {
                            *status = Some(not_found_message(kind));
                        }
                        let cl = catchlight_for(tool, at, p, &to_source, dims);
                        let e = eye_tool::place(found, at, kind, tool, cl);
                        s.edits.eyes.push(e);
                        select(s, tool, s.edits.eyes.len() - 1);
                        changed = true;
                    }
                }
            }
        }

        // ---- drags ----
        if tool_drag_started(resp) {
            if let Some(p) = resp.interact_pointer_pos() {
                s.eye_drag = match eye_tool::hit_test(&s.edits.eyes, to_source(p), aspect, min_grab) {
                    Some((i, g)) => {
                        select(s, tool, i);
                        s.suppressed_eye = Some(i);
                        s.request(s.view_edits(Tool::Retouch), s.render_max_dim(), s.requested_region);
                        EyeDrag::Editing(i, g)
                    }
                    None => EyeDrag::Sizing { at: p, to: p },
                };
            }
        }
        if tool_drag(resp) {
            if let Some(p) = resp.interact_pointer_pos() {
                match s.eye_drag {
                    EyeDrag::Sizing { at, .. } => s.eye_drag = EyeDrag::Sizing { at, to: p },
                    EyeDrag::Editing(i, g) => {
                        let now = to_source(p);
                        let was = to_source(p - resp.drag_delta());
                        if let Some(e) = s.edits.eyes.get_mut(i) {
                            match g {
                                Grab::Move => {
                                    e.centre[0] += now[0] - was[0];
                                    e.centre[1] += now[1] - was[1];
                                }
                                Grab::Resize => {
                                    e.radius = rustroom_app::spot_tool::drag_radius(e.centre, now, aspect);
                                    tool.radius = e.radius;
                                }
                            }
                        }
                    }
                    EyeDrag::None => {}
                }
            }
        }
        if resp.drag_stopped() {
            let live = resp.interact_pointer_pos();
            match s.eye_drag {
                EyeDrag::Sizing { at, to } if (live.unwrap_or(to) - at).length() > CLICK_SLOP => {
                    let c = to_source(at);
                    tool.radius = rustroom_app::spot_tool::drag_radius(c, to_source(live.unwrap_or(to)), aspect);
                    let cl = catchlight_for(tool, c, at, &to_source, dims);
                    s.edits.eyes.push(eye_tool::place(Detection::NotFound, c, kind, tool, cl));
                    select(s, tool, s.edits.eyes.len() - 1);
                    changed = true;
                }
                EyeDrag::Editing(..) => changed = true,
                _ => {}
            }
            if s.eye_drag != EyeDrag::None {
                s.suppressed_eye = None;
                s.request(s.view_edits(Tool::Retouch), s.render_max_dim(), s.requested_region);
                s.eye_drag = EyeDrag::None;
            }
        }
    }
    let changed = changed && s.history.push(s.edits.clone());

    // ---- drawing ----
    let painter = ui.painter_at(canvas_geom::overlay_clip(img_rect));
    for (i, e) in s.edits.eyes.iter().enumerate() {
        let sel = s.selected_eye == Some(i);
        let (halo, mark) = if sel { (140, 255) } else { (70, 110) };
        let c = to_screen(e.centre);
        let r = e.radius * map.radius_scale_at(e.centre);
        for (w, col) in [(3.0, egui::Color32::from_black_alpha(halo)), (1.0, egui::Color32::from_white_alpha(mark))] {
            circle(&painter, c, r, egui::Stroke::new(w, col));
        }
        // A small centre dot distinguishes an eye from a patch outline at a glance.
        painter.circle_filled(c, 1.5, egui::Color32::from_white_alpha(mark));
    }
    if let EyeDrag::Sizing { at, to } = s.eye_drag {
        if (to - at).length() > CLICK_SLOP {
            let r = rustroom_app::spot_tool::drag_radius(to_source(at), to_source(to), aspect)
                * map.radius_scale_at(to_source(at));
            circle(&painter, at, r, egui::Stroke::new(1.5, egui::Color32::from_white_alpha(230)));
        }
    }
    changed
}

/// Select eye `i`, clear the spot selection, and let the tool show that eye's kind.
fn select(s: &mut Session, tool: &mut EyeTool, i: usize) {
    s.selected_eye = Some(i);
    s.selected_spot = None;
    if let Some(e) = s.edits.eyes.get(i) {
        tool.active = Some(e.kind);
    }
}

/// The pet-eye default catchlight: toward the point 10 screen points up-left of the click, mapped
/// into source space — so it is up-left **as displayed**, whatever the orientation.
fn catchlight_for(
    tool: &EyeTool,
    centre: [f32; 2],
    screen: egui::Pos2,
    to_source: &dyn Fn(egui::Pos2) -> [f32; 2],
    dims: (usize, usize),
) -> Option<[f32; 2]> {
    tool.catchlight.then(|| {
        eye_tool::catchlight_offset(centre, to_source(screen + egui::vec2(-10.0, -10.0)), dims)
    })
}

fn not_found_message(kind: EyeKind) -> String {
    let what = match kind {
        EyeKind::Red => "red pupil",
        EyeKind::Pet => "bright eyeshine",
    };
    format!("No {what} found near the click — placed a circle; drag its edge to resize.")
}

fn circle(painter: &egui::Painter, c: egui::Pos2, r: f32, stroke: egui::Stroke) {
    let pts: Vec<egui::Pos2> = (0..CIRCLE_SEGMENTS)
        .map(|k| {
            let t = k as f32 / CIRCLE_SEGMENTS as f32 * std::f32::consts::TAU;
            c + egui::vec2(r * t.cos(), r * t.sin())
        })
        .collect();
    painter.add(egui::Shape::closed_line(pts, stroke));
}
```
