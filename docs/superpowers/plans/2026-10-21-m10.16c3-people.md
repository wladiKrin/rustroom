> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Retouch's Remove mode finds the people in a photo (YOLOX-m, whole frame and tiles), shows each one's
outline (MobileSAM), and removes the ones ticked with c1's fill — each its own removal, one undo step.

**Architecture:** `ai_mask::people` runs YOLOX-m on an upright image. `rustroom_app::people_tool` (lib, no egui) turns

**Tech Stack:** Rust, `tract-onnx` 0.23, egui 0.35, the existing `ai_mask::Sam` and LaMa.

```rust
//! People found by YOLOX-m (M10.16c3) — Apache-2.0, Megvii's own `0.1.1rc0` export: the whole frame at 640 and
//! 640² tiles over it, merged. The boxes only; the app outlines each with MobileSAM and maps it to the decode.
//!
//! ⚠ The export's recipe, measured in the spike (`examples/people_probe.rs`): BGR 0–255, no normalisation, padded
//! with 114 at the right and bottom; the output `[1, 8400, 85]` raw — per stride (8, 16, 32) a grid of offsets and
//! log sizes, then objectness and 80 class scores already through a sigmoid. Person is class 0.

use crate::infer::InferError;
use image::imageops::{self, FilterType};
use image::RgbImage;
use tract_onnx::prelude::*;

/// The network's square.
pub const SIDE: u32 = 640;
/// The tiled pass: the image at this long side, in squares overlapping by [`OVERLAP`].
pub const TILED_LONG: u32 = 1920;
pub const OVERLAP: u32 = 160;
/// A person below this score is not offered — the demo's own drawing threshold is 0.3.
pub const SCORE: f32 = 0.35;
const NMS: f32 = 0.45;
const CONTAINED: f32 = 0.7;
const ANCHORS: usize = 8400;
const ROW: usize = 85;

#[derive(Debug, Clone, Copy, PartialEq)]
pub struct Found {
    /// `[x0, y0, x1, y1]`, normalised over the image handed to [`People::find`].
    pub bbox: [f32; 4],
    pub score: f32,
}

type Plan = std::sync::Arc<TypedRunnableModel>;

fn err(e: impl std::fmt::Display) -> InferError {
    InferError::Model(e.to_string())
}

/// YOLOX-m, loaded and optimised once.
pub struct People {
    plan: Plan,
}

impl People {
    pub fn load(path: &std::path::Path) -> Result<Self, InferError> {
        let plan = tract_onnx::onnx()
            .model_for_path(path)
            .map_err(err)?
            .with_input_fact(0, f32::fact([1, 3, SIDE as usize, SIDE as usize]).into())
            .map_err(err)?
            .into_optimized()
            .map_err(err)?
            .into_runnable()
            .map_err(err)?;
        Ok(Self { plan })
    }

    /// The people in `img` (upright): the whole frame, then the tiles — `progress` told after each square.
    pub fn find(&self, img: &RgbImage, progress: &dyn Fn(f32)) -> Result<Vec<Found>, InferError> {
        let whole_sq = resize_long(img, SIDE);
        let (sw, sh) = (whole_sq.width() as f32, whole_sq.height() as f32);
        let whole = nms(self
            .square(&whole_sq)?
            .into_iter()
            .map(|(b, score)| Found { bbox: norm(b, [0.0, 0.0], sw, sh), score })
            .collect());
        let work = resize_long(img, TILED_LONG);
        let (w, h) = (work.width(), work.height());
        let (xs, ys) = (offsets(w), offsets(h));
        let total = (1 + xs.len() * ys.len()) as f32;
        progress(1.0 / total);
        let mut tiles = Vec::new();
        let mut done = 1.0;
        for &oy in &ys {
            for &ox in &xs {
                let size = [SIDE.min(w - ox), SIDE.min(h - oy)];
                let tile = imageops::crop_imm(&work, ox, oy, size[0], size[1]).to_image();
                for (b, score) in self.square(&tile)? {
                    if !cut(b, [ox, oy], size, w, h) {
                        tiles.push(Found {
                            bbox: norm(b, [ox as f32, oy as f32], w as f32, h as f32),
                            score,
                        });
                    }
                }
                done += 1.0;
                progress(done / total);
            }
        }
        Ok(merge(whole, nms(tiles)))
    }

    /// One square (at most 640², top-left) through the network: the people in its pixels.
    fn square(&self, sq: &RgbImage) -> Result<Vec<([f32; 4], f32)>, InferError> {
        let s = SIDE as usize;
        let n = s * s;
        let mut input = vec![114f32; 3 * n];
        for (x, y, p) in sq.enumerate_pixels() {
            let i = y as usize * s + x as usize;
            input[i] = p[2] as f32;
            input[n + i] = p[1] as f32;
            input[2 * n + i] = p[0] as f32;
        }
        let t = Tensor::from_shape(&[1, 3, s, s], &input).map_err(err)?;
        let out = self.plan.run(tvec!(t.into())).map_err(err)?;
        let o = out[0].to_plain_array_view::<f32>().map_err(err)?;
        if o.shape() != [1, ANCHORS, ROW] {
            return Err(err(format!("YOLOX answered {:?}", o.shape())));
        }
        let o = o.as_slice().ok_or_else(|| err("a non-contiguous output"))?;
        if o.iter().any(|v| !v.is_finite()) {
            return Err(err("a non-finite output"));
        }
        Ok(decode(o))
    }
}

/// The people in one output, in the square's pixels, scoring at least [`SCORE`].
pub fn decode(o: &[f32]) -> Vec<([f32; 4], f32)> {
    let mut found = Vec::new();
    let mut k = 0;
    for stride in [8usize, 16, 32] {
        let g = SIDE as usize / stride;
        for gy in 0..g {
            for gx in 0..g {
                let r = &o[k * ROW..(k + 1) * ROW];
                k += 1;
                let score = r[4] * r[5];
                if score < SCORE {
                    continue;
                }
                let st = stride as f32;
                let (cx, cy) = ((r[0] + gx as f32) * st, (r[1] + gy as f32) * st);
                let (w, h) = (r[2].exp() * st, r[3].exp() * st);
                found.push(([cx - w / 2.0, cy - h / 2.0, cx + w / 2.0, cy + h / 2.0], score));
            }
        }
    }
    found
}

fn area(b: [f32; 4]) -> f32 {
    (b[2] - b[0]).max(0.0) * (b[3] - b[1]).max(0.0)
}

fn overlap(a: [f32; 4], b: [f32; 4]) -> f32 {
    area([a[0].max(b[0]), a[1].max(b[1]), a[2].min(b[2]), a[3].min(b[3])])
}

/// Intersection over union, and the share of the smaller box inside the other.
fn iou_and_containment(a: [f32; 4], b: [f32; 4]) -> (f32, f32) {
    let i = overlap(a, b);
    let u = area(a) + area(b) - i;
    (i / u.max(1e-9), i / area(a).min(area(b)).max(1e-9))
}

/// Strongest first; a box overlapping a kept one by IoU over 0.45, or mostly inside it, goes.
pub fn nms(mut v: Vec<Found>) -> Vec<Found> {
    v.sort_by(|a, b| b.score.total_cmp(&a.score));
    let mut keep: Vec<Found> = Vec::new();
    for f in v {
        if keep.iter().all(|k| {
            let (iou, inside) = iou_and_containment(k.bbox, f.bbox);
            iou <= NMS && inside <= CONTAINED
        }) {
            keep.push(f);
        }
    }
    keep
}

/// The squares' left (or top) edges along `len`: from 0, a square every `SIDE − OVERLAP`, the last flush with
/// the end.
pub fn offsets(len: u32) -> Vec<u32> {
    if len <= SIDE {
        return vec![0];
    }
    let mut v: Vec<u32> = (0..)
        .map(|k| k * (SIDE - OVERLAP))
        .take_while(|&o| o + SIDE < len)
        .collect();
    v.push(len - SIDE);
    v
}

/// Whether box `b` (the tile's pixels) of the tile at `at`, `size` big, in a `w × h` image, touches an edge the
/// image goes on past — the next tile holds that person whole.
fn cut(b: [f32; 4], at: [u32; 2], size: [u32; 2], w: u32, h: u32) -> bool {
    (b[0] < 3.0 && at[0] > 0)
        || (b[1] < 3.0 && at[1] > 0)
        || (b[2] > size[0] as f32 - 3.0 && at[0] + size[0] < w)
        || (b[3] > size[1] as f32 - 3.0 && at[1] + size[1] < h)
}

/// The whole frame's finds, then each tile's that none of them covers.
pub fn merge(whole: Vec<Found>, tiles: Vec<Found>) -> Vec<Found> {
    let mut all = whole.clone();
    for t in tiles {
        let seen = whole.iter().any(|p| {
            let (iou, inside) = iou_and_containment(p.bbox, t.bbox);
            iou > 0.3 || inside > 0.6
        });
        if !seen {
            all.push(t);
        }
    }
    all
}

fn norm(b: [f32; 4], at: [f32; 2], w: f32, h: f32) -> [f32; 4] {
    [
        (b[0] + at[0]) / w,
        (b[1] + at[1]) / h,
        (b[2] + at[0]) / w,
        (b[3] + at[1]) / h,
    ]
    .map(|v| v.clamp(0.0, 1.0))
}

fn resize_long(img: &RgbImage, long: u32) -> RgbImage {
    let r = long as f32 / img.width().max(img.height()).max(1) as f32;
    imageops::resize(
        img,
        ((img.width() as f32 * r).round() as u32).max(1),
        ((img.height() as f32 * r).round() as u32).max(1),
        FilterType::Triangle,
    )
}
```

```rust
/// A `side²` plane of `n` channels turned `turns` clockwise quarter turns.
fn turn_square(v: &[f32], side: usize, n: usize, turns: u8) -> Vec<f32> {
    let mut cur = v.to_vec();
    for _ in 0..turns % 4 {
        let mut next = vec![0f32; cur.len()];
        for y in 0..side {
            for x in 0..side {
                // Clockwise: (x, y) → (side − 1 − y, x).
                let (nx, ny) = (side - 1 - y, x);
                next[(ny * side + nx) * n..(ny * side + nx + 1) * n]
                    .copy_from_slice(&cur[(y * side + x) * n..(y * side + x + 1) * n]);
            }
        }
        cur = next;
    }
    cur
}

/// A normalised box turned `turns` clockwise quarter turns — its two corners turned, re-ordered.
fn turn_box(b: [f32; 4], turns: u8) -> [f32; 4] {
    let a = crate::ai_worker::turn_point([b[0], b[1]], turns);
    let c = crate::ai_worker::turn_point([b[2], b[3]], turns);
    [a[0].min(c[0]), a[1].min(c[1]), a[0].max(c[0]), a[1].max(c[1])]
}
```

```rust
//! Find people (M10.16c3), outside any thread or window: a search over the upright photo — YOLOX-m's boxes,
//! MobileSAM's outline of each — handed back in DECODE space, the pre-tick, and the removals the ticked ones become.

use crate::remove_tool::Area;
use std::path::PathBuf;
use std::sync::Arc;

/// A person this tall — a share of the upright frame's height — or taller is the photo's subject: found, shown,
/// never pre-ticked. Right for the subject on all seven of the spike's sheets.
pub const SUBJECT: f32 = 0.35;
/// The side a person's outline is kept at for the canvas — a preview; the fill makes its own at 512.
pub const AREA_SIDE: usize = 128;
/// The outline's rectangle: the box grown by this share of its size a side, so the shading has room.
const AREA_MARGIN: f32 = 0.1;

/// One person found, in decode space.
#[derive(Debug, Clone)]
pub struct Person {

    pub score: f32,

    pub tall: f32,

/// One person as a finder answers: in the upright image, their outline over the whole of it.
pub struct UprightPerson {
    pub bbox: [f32; 4],
    pub score: f32,
    pub mask: ai_mask::ProbMap,
}

/// YOLOX-m from `models`, then MobileSAM embedded once and prompted with each box — the people's share of the
/// progress three quarters, the outlines' the rest.
pub fn real_finder(models: PathBuf) -> Finder {
    Box::new(move |img, progress| {
        let people = ai_mask::people::People::load(&models.join(ai_mask::ModelId::YoloxM.spec().file))
            .map_err(|e| e.to_string())?;
        let found = people.find(img, &|p| progress(0.75 * p)).map_err(|e| e.to_string())?;
        let [enc, dec] = ai_mask::registry::CLICK_MODELS.map(|m| models.join(m.spec().file));
        let mut sam = ai_mask::Sam::new(&enc, &dec);
        let e = sam.embed(img).map_err(|e| e.to_string())?;
        let n = found.len().max(1) as f32;
        let mut out = Vec::new();
        for (k, f) in found.into_iter().enumerate() {
            let mask = sam.prompt(&e, &[], Some(f.bbox)).map_err(|e| e.to_string())?;
            out.push(UprightPerson { bbox: f.bbox, score: f.score, mask });
            progress(0.75 + 0.25 * (k + 1) as f32 / n);
        }
        Ok(out)
    })
}

/// `p` in decode space: the photo was turned `turns` clockwise quarter turns to upright, so each point is turned
/// back by the rest of the circle.
pub fn to_decode(p: UprightPerson, turns: u8) -> Person {
    let back = (4 - turns % 4) % 4;
    let [x0, y0, x1, y1] = p.bbox;
    let (gx, gy) = (AREA_MARGIN * (x1 - x0), AREA_MARGIN * (y1 - y0));
    let up_rect = [(x0 - gx).max(0.0), (y0 - gy).max(0.0), (x1 + gx).min(1.0), (y1 + gy).min(1.0)];
    let turn_box = |b: [f32; 4]| {
        let a = crate::ai_worker::turn_point([b[0], b[1]], back);
        let c = crate::ai_worker::turn_point([b[2], b[3]], back);
        [a[0].min(c[0]), a[1].min(c[1]), a[0].max(c[0]), a[1].max(c[1])]
    };
    let rect = turn_box(up_rect);
    // Each decode-side texel of the area asks the upright mask at the point it turns to.
    let s = AREA_SIDE;
    let alpha = (0..s * s)
        .map(|i| {
            let (u, v) = (((i % s) as f32 + 0.5) / s as f32, ((i / s) as f32 + 0.5) / s as f32);
            let d = [rect[0] + u * (rect[2] - rect[0]), rect[1] + v * (rect[3] - rect[1])];
            let up = crate::ai_worker::turn_point(d, turns % 4);
            p.mask.at(up[0], up[1])
        })
        .collect();
    Person {
        bbox: turn_box(p.bbox),
        score: p.score,
        tall: y1 - y0,
        area: Arc::new(Area { rect, side: s, alpha }),
    }
}

pub fn pre_ticked(people: &[Person]) -> Vec<bool> {
    people.iter().map(|p| p.tall < SUBJECT).collect()
}

/// One removal per ticked person: their box, no stroke, the full Opacity.
pub fn removals(people: &[Person], ticked: &[bool]) -> Vec<edit_model::Removal> {
    people
        .iter()
        .zip(ticked)
        .filter(|(_, &t)| t)
        .map(|(p, _)| edit_model::Removal {
            points: Vec::new(),
            radius: 0.0,
            detect: false,
            variant: 0,
            opacity: 100.0,
            person: Some(p.bbox),
        })
        .collect()
}

pub fn summary(n: usize, ticked: usize) -> String {
    match n {
        0 => "No people found".to_string(),
        1 => format!("1 person · {ticked} to remove"),
        n => format!("{n} people · {ticked} to remove"),
    }
}

pub fn remove_label(ticked: usize) -> String {
    match ticked {
        1 => "Remove 1 person".to_string(),
        n => format!("Remove {n} people"),
    }
}
```

```rust
//! Find people's job (M10.16c3): a press starts the search in the background — once the models are downloaded and
//! the photo has loaded — and the people it finds land on the session, pre-ticked. The wiring only; every
//! decision is `rustroom_app::people_tool`'s.
//!
//! ⚠ Pumped by Develop's body, `dust_flow`'s reason.

/// Below this long side the decode is the stand-in shown while the photo loads.
const LOADED: usize = 1500;
const REDRAW: std::time::Duration = std::time::Duration::from_millis(50);
/// The long side the people are found at — the tiles' own (`ai_mask::people::TILED_LONG`).
const INPUT_LONG: usize = ai_mask::people::TILED_LONG as usize;

/// The upright image the search reads: the decode's positive, sRGB-encoded — the AI masks' own input.
pub(crate) fn people_input(buffer: &ImageBuf, turns: u8, negative: &edit_model::Negative) -> image::RgbImage {
    rustroom_app::ai_worker::model_input(buffer, INPUT_LONG, turns, negative)
}

impl RustroomApp {
    pub(crate) fn people_models(&self) -> Option<PathBuf> {
        match &self.ai.dirs {
            Some((_, models)) => Some(models.clone()),
            None => rustroom_app::ai_cache::models_dir(),
        }
    }

    /// Once per Develop frame. `false` always — finding people edits nothing; Remove N people is the panel's step.
    pub(crate) fn pump_people(&mut self, ctx: &egui::Context) -> bool {
        let models = self.people_models();
        let missing: Vec<ai_mask::ModelId> = /* [YoloxM] + CLICK_MODELS not `present` in `models` */;
        let downloading = self.ai.download.is_some() || self.modal == Modal::AiDownload;
        let Some(s) = self.session.as_mut() else { return false };
        let p = &mut s.people;
        if std::mem::take(&mut p.pressed) {
            p.said = None;
            p.people.clear();
            p.ticked.clear();
            p.waiting = Some(Wait::Photo);
            if !missing.is_empty() {
                p.waiting = Some(Wait::Models);
                self.ai.proposed = missing.clone();
                self.modal = Modal::AiDownload;
            }
        }
        match p.waiting {
            Some(Wait::Models) if missing.is_empty() => p.waiting = Some(Wait::Photo),
            Some(Wait::Models) if !downloading => {
                p.waiting = None;
                p.said = Some("Find people needs its models".to_string());
            }
            _ => {}
        }
        if matches!(p.waiting, Some(Wait::Photo))
            && !s.buffer_is_ghost
            && s.buffer.width().max(s.buffer.height()) >= LOADED
        {
            let turns = s.edits.geometry.orientation.rotate.turns();
            let img = Arc::new(people_input(&s.buffer, turns, &s.edits.negative));
            let models = models.expect("present above");
            s.people.waiting = None;
            s.people.search = Some(Search::spawn(img, turns, people_tool::real_finder(models)));
        }
        let p = &mut s.people;
        if p.waiting.is_some() {
            ctx.request_repaint_after(REDRAW);
        }
        let Some(search) = &p.search else { return false };
        let Some(got) = search.poll() else {
            ctx.request_repaint_after(REDRAW);
            return false;
        };
        p.search = None;
        match got {
            Ok(people) => {
                p.ticked = people_tool::pre_ticked(&people);
                p.said = Some(people_tool::summary(people.len(), p.ticked.iter().filter(|t| **t).count()));
                p.people = people;
            }
            Err(e) => p.said = Some(format!("Could not find people: {e}")),
        }
        false
    }
}
```

```rust
const BUTTON_HOVER: &str = "Finds the people in the photo so you can pick the ones to remove";
const LOOKING_HOVER: &str = "Each person found is outlined on the photo, the ones ticked in the accent";
const EVERYONE_HOVER: &str = "Ticks every person found, or none";
const REMOVE_HOVER: &str = "Removes each ticked person, filled from their surroundings";
const CANCEL_HOVER: &str = "Stops showing the people found without removing anyone";

pub(super) fn show(ui: &mut egui::Ui, s: &mut Session, ed: &mut Edited) {
    ui.add_space(6.0);
    let looking = s.people.search.is_some();
    if ui.add_enabled(!looking, egui::Button::new("Find people")).hover(BUTTON_HOVER).clicked() {
        s.people.pressed = true;
    }
    if let Some(search) = &s.people.search {
        super::progress_line::show(ui, "Looking for people", search.progress(), LOOKING_HOVER);
        return;
    }
    match s.people.waiting {
        Some(Wait::Photo) => return theme::hint(ui, "Waits for the photo to load"),
        Some(Wait::Models) => return theme::hint(ui, "Waits for its models to download"),
        None => {}
    }
    if s.people.people.is_empty() {
        if let Some(said) = &s.people.said { theme::hint(ui, said); }
        return;
    }
    let n = s.people.people.len();
    let ticked = s.people.ticked.iter().filter(|t| **t).count();
    theme::hint(ui, &people_tool::summary(n, ticked));
    let mut all = ticked == n;
    if ui.add(egui::Checkbox::new(&mut all, "Everyone").indeterminate(ticked > 0 && ticked < n))
        .hover(EVERYONE_HOVER).changed()
    {
        s.people.ticked.iter_mut().for_each(|t| *t = all);
    }
    ui.horizontal(|ui| {
        if ui.add_enabled(ticked > 0, egui::Button::new(people_tool::remove_label(ticked)))
            .hover(REMOVE_HOVER).clicked()
        {
            let new = people_tool::removals(&s.people.people, &s.people.ticked);
            s.edits.removals.extend(new);
            s.removal.sync(s.edits.removals.len());
            s.removal.selected = None;
            s.people.people.clear();
            s.people.ticked.clear();
            s.people.said = None;
            ed.discrete();
        }
        if ui.button("Cancel").hover(CANCEL_HOVER).clicked() {
            s.people.people.clear();
            s.people.ticked.clear();
            s.people.said = None;
        }
    });
}
```
